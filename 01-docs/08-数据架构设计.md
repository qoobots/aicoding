## 1. 数据存储策略

### 1.1 混合存储方案

AI编程平台采用**混合存储方案**,根据数据特点选择最适合的存储方式:

| 数据类型       | 存储方式 | 理由                     |
| -------------- | -------- | ------------------------ |
| 项目元数据     | SQLite   | 需要查询、事务、复杂关联 |
| 生成的代码文件 | 文件系统 | 大文件存储,便于版本控制  |
| 模板文件       | 文件系统 | 便于编辑和管理           |
| AI生成记录     | SQLite   | 需要查询、统计、历史追溯 |
| 用户偏好设置   | SQLite   | 频繁读写,需要持久化      |
| 缓存数据       | 文件系统 | 临时数据,可随时删除      |

### 2.2 数据表设计

#### 2.2.1 项目表 (projects)

```sql
CREATE TABLE projects (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  template_type TEXT NOT NULL,  -- spring-boot, spring-cloud, etc.
  status TEXT DEFAULT 'active',  -- active, archived, deleted
  config TEXT,                    -- JSON配置
  project_path TEXT NOT NULL,     -- 项目文件路径
  created_at INTEGER NOT NULL,   -- Unix时间戳
  updated_at INTEGER NOT NULL
);

CREATE INDEX idx_projects_status ON projects(status);
CREATE INDEX idx_projects_template_type ON projects(template_type);
CREATE INDEX idx_projects_created_at ON projects(created_at);
```

**字段说明**:

- `id`: 项目唯一标识 (UUID)
- `name`: 项目名称
- `description`: 项目描述
- `template_type`: 模板类型
- `status`: 项目状态
- `config`: 项目配置 (JSON格式,存储包名、端口等)
- `project_path`: 项目文件存储路径
- `created_at`: 创建时间
- `updated_at`: 更新时间

#### 2.2.2 代码文件表 (code_files)

```sql
CREATE TABLE code_files (
  id TEXT PRIMARY KEY,
  project_id TEXT NOT NULL,
  file_path TEXT NOT NULL,         -- 相对路径
  content TEXT,                    -- 小文件内容直接存储
  file_hash TEXT NOT NULL,         -- 文件MD5哈希
  language TEXT NOT NULL,          -- 编程语言
  file_size INTEGER NOT NULL,      -- 文件大小(字节)
  created_at INTEGER NOT NULL,
  updated_at INTEGER NOT NULL,
  UNIQUE(project_id, file_path),
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE
);

CREATE INDEX idx_code_files_project_id ON code_files(project_id);
CREATE INDEX idx_code_files_hash ON code_files(file_hash);
CREATE INDEX idx_code_files_language ON code_files(language);
```

**字段说明**:

- `id`: 文件唯一标识 (UUID)
- `project_id`: 所属项目ID
- `file_path`: 文件相对路径
- `content`: 文件内容 (小文件<10KB直接存储)
- `file_hash`: 文件MD5哈希 (用于去重)
- `language`: 编程语言 (java, typescript, vue等)
- `file_size`: 文件大小 (字节)

### 5.2 数据备份策略

```typescript
// 数据备份服务

@Injectable()
export class BackupService {
  async backupDatabase(): Promise<string> {
    const timestamp = Date.now();
    const backupPath = path.join(this.backupDir, `backup-${timestamp}.db`);

    // 复制SQLite数据库
    await fs.promises.copyFile(this.dbPath, backupPath);

    // 删除7天前的备份
    await this.cleanOldBackups(7);

    return backupPath;
  }

  async restoreDatabase(backupPath: string): Promise<void> {
    // 停止所有数据库连接
    await this.closeAllConnections();

    // 恢复备份
    await fs.promises.copyFile(backupPath, this.dbPath);

    // 重新连接数据库
    await this.reconnectDatabase();
  }

  async cleanOldBackups(days: number): Promise<void> {
    const cutoffTime = Date.now() - days * 24 * 60 * 60 * 1000;

    const files = await fs.promises.readdir(this.backupDir);
    for (const file of files) {
      const filePath = path.join(this.backupDir, file);
      const stats = await fs.promises.stat(filePath);

      if (stats.mtimeMs < cutoffTime) {
        await fs.promises.unlink(filePath);
      }
    }
  }
}
```

## 7. 数据安全

### 7.1 敏感数据加密

```typescript
import * as crypto from 'crypto';

@Injectable()
export class EncryptionService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly key = crypto.scryptSync(process.env.ENCRYPTION_KEY || 'default-key', 'salt', 32);

  encrypt(text: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv(this.algorithm, this.key, iv);

    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    const authTag = cipher.getAuthTag();

    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted}`;
  }

  decrypt(encrypted: string): string {
    const [ivHex, authTagHex, encryptedData] = encrypted.split(':');
    const iv = Buffer.from(ivHex, 'hex');
    const authTag = Buffer.from(authTagHex, 'hex');

    const decipher = crypto.createDecipheriv(this.algorithm, this.key, iv);
    decipher.setAuthTag(authTag);

    let decrypted = decipher.update(encryptedData, 'hex', 'utf8');
    decrypted += decipher.final('utf8');

    return decrypted;
  }
}
```

### 7.2 API密钥管理

```typescript
// AI API密钥加密存储

@Injectable()
export class AIService {
  constructor(
    private readonly encryptionService: EncryptionService,
    private readonly userPreferencesService: UserPreferencesService
  ) {}

  async saveApiKey(apiKey: string): Promise<void> {
    const encrypted = this.encryptionService.encrypt(apiKey);
    await this.userPreferencesService.set('ai.apiKey', encrypted);
  }

  async getApiKey(): Promise<string> {
    const encrypted = await this.userPreferencesService.get('ai.apiKey');
    if (!encrypted) return null;
    return this.encryptionService.decrypt(encrypted);
  }
}
```

### 8.3 缓存策略

```typescript
import { Cache } from 'cache-manager';

@Injectable()
export class ProjectService {
  constructor(
    @Inject(CACHE_MANAGER) private cacheManager: Cache,
    private readonly projectRepo: Repository<Project>
  ) {}

  async getProject(id: string): Promise<Project> {
    // 先从缓存读取
    const cacheKey = `project:${id}`;
    const cached = await this.cacheManager.get<Project>(cacheKey);

    if (cached) {
      return cached;
    }

    // 缓存未命中,从数据库读取
    const project = await this.projectRepo.findOne({ where: { id } });

    // 写入缓存 (1小时过期)
    await this.cacheManager.set(cacheKey, project, 3600);

    return project;
  }
}
```
