## 1. 交互设计原则

### 1.1 核心原则

- **简洁高效**: 最少点击次数完成操作
- **即时反馈**: 所有操作都有明确反馈
- **容错友好**: 支持撤销和错误恢复
- **一致性**: 保持交互模式统一
- **可预测**: 符合用户心理预期

### 1.2 设计目标

| 指标         | 目标值      | 说明               |
| ------------ | ----------- | ------------------ |
| 任务完成时间 | < 3分钟     | 完成常见任务的时间 |
| 错误率       | < 5%        | 用户操作错误率     |
| 学习曲线     | 5分钟内上手 | 新用户快速上手     |
| 满意度       | > 90%       | 用户满意度评分     |

```typescript
// 使用Pinia管理加载状态

export const useUIStore = defineStore('ui', () => {
  const loading = ref(false);
  const loadingMessage = ref('');
  const blocking = ref(false);

  function startLoading(message = '加载中...', block = false) {
    loading.value = true;
    loadingMessage.value = message;
    blocking.value = block;

    if (block) {
      document.body.style.overflow = 'hidden';
    }
  }

  function stopLoading() {
    loading.value = false;
    loadingMessage.value = '';
    blocking.value = false;
    document.body.style.overflow = '';
  }

  return {
    loading,
    loadingMessage,
    blocking,
    startLoading,
    stopLoading,
  };
});
```

## 7. 响应式设计

### 7.1 断点定义

```typescript
// 断点配置
export const breakpoints = {
  xs: 0, // 手机
  sm: 640, // 平板竖屏
  md: 768, // 平板横屏
  lg: 1024, // 桌面
  xl: 1280, // 大桌面
  '2xl': 1536, // 超大屏
};

// 响应式Hook
export function useBreakpoint() {
  const currentBreakpoint = ref('lg');

  const updateBreakpoint = () => {
    const width = window.innerWidth;

    if (width < breakpoints.sm) {
      currentBreakpoint.value = 'xs';
    } else if (width < breakpoints.md) {
      currentBreakpoint.value = 'sm';
    } else if (width < breakpoints.lg) {
      currentBreakpoint.value = 'md';
    } else if (width < breakpoints.xl) {
      currentBreakpoint.value = 'lg';
    } else if (width < breakpoints['2xl']) {
      currentBreakpoint.value = 'xl';
    } else {
      currentBreakpoint.value = '2xl';
    }
  };

  onMounted(() => {
    updateBreakpoint();
    window.addEventListener('resize', updateBreakpoint);
  });

  onUnmounted(() => {
    window.removeEventListener('resize', updateBreakpoint);
  });

  return {
    currentBreakpoint,
    isMobile: computed(() => ['xs', 'sm'].includes(currentBreakpoint.value)),
    isTablet: computed(() => currentBreakpoint.value === 'md'),
    isDesktop: computed(() => ['lg', 'xl', '2xl'].includes(currentBreakpoint.value)),
  };
}
```

```typescript
// 重要操作二次确认

export async function confirmAction(
  message: string,
  options?: {
    title?: string;
    confirmText?: string;
    cancelText?: string;
    type?: 'warning' | 'danger';
  }
): Promise<boolean> {
  const {
    title = '确认操作',
    confirmText = '确认',
    cancelText = '取消',
    type = 'warning',
  } = options || {};

  return new Promise((resolve) => {
    Dialog.confirm({
      title,
      content: message,
      confirmBtn: {
        content: confirmText,
        theme: type,
      },
      cancelBtn: {
        content: cancelText,
      },
      onConfirm: () => resolve(true),
      onCancel: () => resolve(false),
    });
  });
}

// 使用示例
async function handleDeleteProject(project: Project) {
  const confirmed = await confirmAction(`确定要删除项目"${project.name}"吗?此操作不可恢复。`, {
    title: '删除项目',
    type: 'danger',
  });

  if (confirmed) {
    await projectService.delete(project.id);
    MessagePlugin.success('项目已删除');
  }
}
```

### 8.2 撤销/重做

```typescript
// 撤销重做栈管理

export function useUndoRedo<T>() {
  const history = ref<T[]>([]);
  const currentIndex = ref(-1);
  const maxHistorySize = 50;

  function pushState(state: T) {
    // 移除当前位置之后的状态
    history.value = history.value.slice(0, currentIndex.value + 1);

    // 添加新状态
    history.value.push(state);

    // 限制历史记录大小
    if (history.value.length > maxHistorySize) {
      history.value.shift();
    }

    currentIndex.value = history.value.length - 1;
  }

  function undo(): T | null {
    if (currentIndex.value > 0) {
      currentIndex.value--;
      return history.value[currentIndex.value];
    }
    return null;
  }

  function redo(): T | null {
    if (currentIndex.value < history.value.length - 1) {
      currentIndex.value++;
      return history.value[currentIndex.value];
    }
    return null;
  }

  function canUndo(): boolean {
    return currentIndex.value > 0;
  }

  function canRedo(): boolean {
    return currentIndex.value < history.value.length - 1;
  }

  function clear() {
    history.value = [];
    currentIndex.value = -1;
  }

  return {
    history,
    currentIndex,
    pushState,
    undo,
    redo,
    canUndo,
    canRedo,
    clear,
  };
}
```
