### 2.2 Electron主进程核心配置

```typescript
// apps/electron/src/main.ts
import { app, BrowserWindow, ipcMain } from 'electron';
import path from 'path';
import { spawn } from 'child_process';

let nestjsProcess: ChildProcess;
let mainWindow: BrowserWindow | null = null;

// 启动NestJS后端服务
async function startNestJS() {
  const backendPath = path.join(__dirname, '../backend/dist/main.js');

  nestjsProcess = spawn('node', [backendPath], {
    stdio: 'inherit',
    env: {
      ...process.env,
      NODE_ENV: 'production',
    },
  });

  nestjsProcess.on('error', (err) => {
    console.error('NestJS启动失败:', err);
  });

  nestjsProcess.on('exit', (code) => {
    console.log(`NestJS进程退出,代码: ${code}`);
  });

  // 等待后端启动
  await new Promise((resolve) => setTimeout(resolve, 3000));
  console.log('NestJS后端已启动');
}

// 创建主窗口
function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1600,
    height: 1000,
    minWidth: 1200,
    minHeight: 800,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      nodeIntegration: false,
      contextIsolation: true,
      sandbox: true,
    },
    icon: path.join(__dirname, '../assets/icon.png'),
    show: false,
  });

  // 生产环境加载打包后的文件
  if (process.env.NODE_ENV === 'production') {
    mainWindow.loadFile(path.join(__dirname, '../frontend/dist/index.html'));
  } else {
    mainWindow.loadURL('http://localhost:5173');
    mainWindow.webContents.openDevTools();
  }

  mainWindow.once('ready-to-show', () => {
    mainWindow?.show();
  });

  mainWindow.on('closed', () => {
    mainWindow = null;
  });
}

// 应用启动
app.whenReady().then(async () => {
  await startNestJS();
  createWindow();
});

// 应用退出
app.on('quit', () => {
  if (nestjsProcess) {
    nestjsProcess.kill();
    console.log('NestJS后端已关闭');
  }
});
```

### 4.3 Pinia状态管理

```typescript
// apps/frontend/src/stores/project.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Project, ProjectConfig } from '@shared/types';
import { projectService } from '@/services/project.service';

export const useProjectStore = defineStore('project', () => {
  // 状态
  const projects = ref<Project[]>([]);
  const currentProject = ref<Project | null>(null);
  const loading = ref(false);
  const error = ref<string | null>(null);

  // 计算属性
  const activeProjects = computed(() => projects.value.filter((p) => p.status === 'active'));

  const totalProjects = computed(() => projects.value.length);

  // Actions
  async function fetchProjects() {
    loading.value = true;
    error.value = null;

    try {
      const response = await projectService.getProjects();
      projects.value = response.data;
    } catch (e: any) {
      error.value = e.message;
      console.error('获取项目列表失败:', e);
    } finally {
      loading.value = false;
    }
  }

  async function createProject(config: ProjectConfig) {
    loading.value = true;
    error.value = null;

    try {
      const response = await projectService.create(config);
      projects.value.push(response.data);
      return response.data;
    } catch (e: any) {
      error.value = e.message;
      throw e;
    } finally {
      loading.value = false;
    }
  }

  async function deleteProject(id: string) {
    try {
      await projectService.delete(id);
      projects.value = projects.value.filter((p) => p.id !== id);

      if (currentProject.value?.id === id) {
        currentProject.value = null;
      }
    } catch (e: any) {
      error.value = e.message;
      throw e;
    }
  }

  function setCurrentProject(project: Project) {
    currentProject.value = project;
  }

  return {
    // 状态
    projects,
    currentProject,
    loading,
    error,

    // 计算属性
    activeProjects,
    totalProjects,

    // Actions
    fetchProjects,
    createProject,
    deleteProject,
    setCurrentProject,
  };
});
```

```vue
<template>
  <div class="monaco-editor-container" ref="container"></div>

<script setup lang="ts">
import { ref, onMounted, onBeforeUnmount, watch } from 'vue';
import * as monaco from 'monaco-editor';
import type * as Monaco from 'monaco-editor';

const props = defineProps<{
  modelValue: string;
  language?: string;
  theme?: string;
  readOnly?: boolean;
  height?: string;
}>();

const emit = defineEmits<{
  (e: 'update:modelValue', value: string): void;
  (e: 'change', value: string): void;
  (e: 'ready', editor: Monaco.editor.IStandaloneCodeEditor): void;
}>();

const container = ref<HTMLElement>();
let editor: Monaco.editor.IStandaloneCodeEditor | null = null;

onMounted(() => {
  if (!container.value) return;

  editor = monaco.editor.create(container.value, {
    value: props.modelValue,
    language: props.language || 'java',
    theme: props.theme || 'vs-dark',
    readOnly: props.readOnly || false,
    automaticLayout: true,
    fontSize: 14,
    lineNumbers: 'on',
    minimap: { enabled: true },
    scrollBeyondLastLine: false,
    tabSize: 4,
    wordWrap: 'on',
    folding: true,
    foldingStrategy: 'auto',
    showFoldingControls: 'always',
    formatOnPaste: true,
    formatOnType: true,
  });

  // 监听内容变化
  editor.onDidChangeModelContent(() => {
    const value = editor?.getValue() || '';
    emit('update:modelValue', value);
    emit('change', value);
  });

  // 触发ready事件
  emit('ready', editor);
});

// 监听language变化
watch(
  () => props.language,
  (newLanguage) => {
    if (editor && newLanguage) {
      monaco.editor.setModelLanguage(editor.getModel()!, newLanguage);
    }
  }
);

// 监听theme变化
watch(
  () => props.theme,
  (newTheme) => {
    if (editor && newTheme) {
      monaco.editor.setTheme(newTheme);
    }
  }
);

// 监听value变化
watch(
  () => props.modelValue,
  (newValue) => {
    if (editor && newValue !== editor.getValue()) {
      editor.setValue(newValue);
    }
  }
);

onBeforeUnmount(() => {
  editor?.dispose();
});
</script>
```

### 5.2 AI代码补全集成

```typescript
// apps/frontend/src/composables/useAICompletion.ts
import { ref } from 'vue';
import type { Monaco } from 'monaco-editor';

export function useAICompletion(editor: Monaco.editor.IStandaloneCodeEditor) {
  const loading = ref(false);

  async function provideCompletionItems(
    position: Monaco.Position,
    context: Monaco.languages.CompletionContext
  ): Promise<Monaco.languages.CompletionItem[]> {
    loading.value = true;

    try {
      const model = editor.getModel();
      if (!model) return [];

      const lineContent = model.getLineContent(position.lineNumber);
      const cursorText = lineContent.substring(0, position.column - 1);

      // 调用AI服务获取补全建议
      const suggestions = await fetchAISuggestions({
        code: model.getValue(),
        cursorPosition: position,
        context: context,
      });

      return suggestions.map((s) => ({
        label: s.label,
        kind: s.kind,
        insertText: s.insertText,
        detail: s.detail,
        documentation: s.documentation,
        range: s.range,
      }));
    } catch (error) {
      console.error('AI补全失败:', error);
      return [];
    } finally {
      loading.value = false;
    }
  }

  // 注册AI补全提供器
  function registerCompletionProvider(monaco: Monaco) {
    const provider = monaco.languages.registerCompletionItemProvider('java', {
      provideCompletionItems,
    });

    return () => provider.dispose();
  }

  return {
    loading,
    registerCompletionProvider,
  };
}
```

### 6.2 OpenAI提供者实现

```typescript
// apps/backend/src/modules/ai-service/openai.provider.ts
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import OpenAI from 'openai';
import { AIProvider, GenerateOptions } from './ai-provider.interface';

@Injectable()
export class OpenAIProvider implements AIProvider {
  name = 'OpenAI';
  private client: OpenAI;

  constructor(private readonly configService: ConfigService) {
    this.client = new OpenAI({
      apiKey: this.configService.get<string>('OPENAI_API_KEY'),
      baseURL: this.configService.get<string>('OPENAI_BASE_URL'),
    });
  }

  async generateCode(prompt: string, options?: GenerateOptions): Promise<string> {
    const response = await this.client.chat.completions.create({
      model: options?.model || 'gpt-4-turbo-preview',
      messages: [
        {
          role: 'system',
          content: '你是一个专业的代码生成助手,请生成高质量、规范的代码。',
        },
        {
          role: 'user',
          content: prompt,
        },
      ],
      temperature: options?.temperature || 0.7,
      max_tokens: options?.maxTokens || 4096,
    });

    return response.choices[0].message.content || '';
  }

  async reviewCode(code: string): Promise<ReviewResult> {
    const response = await this.client.chat.completions.create({
      model: 'gpt-4-turbo-preview',
      messages: [
        {
          role: 'system',
          content: '你是一个代码审查专家,请审查代码并提供改进建议。',
        },
        {
          role: 'user',
          content: `请审查以下代码:\n\n${code}`,
        },
      ],
      temperature: 0.3,
    });

    // 解析响应并返回结构化结果
    return this.parseReviewResponse(response.choices[0].message.content || '');
  }

  private parseReviewResponse(response: string): ReviewResult {
    // 实现解析逻辑
    return {
      issues: [],
      suggestions: [],
      score: 0,
    };
  }
}
```

```typescript
    Handlebars.registerHelper('camelToSnake', (str: string) => {
      return str.replace(/[A-Z]/g, (letter) => `_${letter.toLowerCase()}`);
    });

    // 注册下划线转驼峰辅助函数
    Handlebars.registerHelper('snakeToCamel', (str: string) => {
      return str.replace(/_([a-z])/g, (_, letter) => letter.toUpperCase());
    });

    // 注册首字母大写辅助函数
    Handlebars.registerHelper('capitalize', (str: string) => {
      return str.charAt(0).toUpperCase() + str.slice(1);
    });

    // 注册日期格式化辅助函数
    Handlebars.registerHelper('dateFormat', (date: Date, format: string) => {
      // 实现日期格式化
      return date.toISOString();
    });
  }

  private async loadTemplates() {
    const templateDir = path.join(process.cwd(), 'data/templates');

    const loadTemplate = async (filePath: string) => {
      const content = await fs.promises.readFile(filePath, 'utf-8');
      const template = Handlebars.compile(content);
      this.templates.set(filePath, template);
    };

    // 加载Spring Boot模板
    await loadTemplate(path.join(templateDir, 'spring-boot/entity.hbs'));
    await loadTemplate(path.join(templateDir, 'spring-boot/controller.hbs'));
    await loadTemplate(path.join(templateDir, 'spring-boot/service.hbs'));
    // 加载更多模板...
  }

  render(templateName: string, data: any): string {
    const template = this.templates.get(templateName);
    if (!template) {
      throw new Error(`模板不存在: ${templateName}`);
    }
    return template(data);
  }
}
```

### 7.2 Java代码生成器

```typescript
// apps/backend/src/modules/code-generator/java-generator.service.ts
import { Injectable } from '@nestjs/common';
import { TemplateEngineService } from './template-engine.service';
import type { EntityConfig, ProjectConfig } from '@shared/types';

@Injectable()
export class JavaGeneratorService {
  constructor(private readonly templateEngine: TemplateEngineService) {}

  async generateEntity(config: EntityConfig): Promise<string> {
    return this.templateEngine.render('spring-boot/entity.hbs', config);
  }

  async generateController(config: EntityConfig): Promise<string> {
    return this.templateEngine.render('spring-boot/controller.hbs', config);
  }

  async generateService(config: EntityConfig): Promise<string> {
    return this.templateEngine.render('spring-boot/service.hbs', config);
  }

  async generateRepository(config: EntityConfig): Promise<string> {
    return this.templateEngine.render('spring-boot/repository.hbs', config);
  }

  async generateProject(projectConfig: ProjectConfig): Promise<Map<string, string>> {
    const files = new Map<string, string>();

    // 生成实体类
    for (const entity of projectConfig.entities) {
      const entityCode = await this.generateEntity(entity);
      const entityPath = `src/main/java/${projectConfig.packageName.replace(/\./g, '/')}/entity/${entity.className}.java`;
      files.set(entityPath, entityCode);

      // 生成Controller
      const controllerCode = await this.generateController(entity);
      const controllerPath = `src/main/java/${projectConfig.packageName.replace(/\./g, '/')}/controller/${entity.className}Controller.java`;
      files.set(controllerPath, controllerCode);

      // 生成Service
      const serviceCode = await this.generateService(entity);
      const servicePath = `src/main/java/${projectConfig.packageName.replace(/\./g, '/')}/service/${entity.className}Service.java`;
      files.set(servicePath, serviceCode);
    }

    return files;
  }
}
```
