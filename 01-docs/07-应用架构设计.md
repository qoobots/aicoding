## 4. 核心应用设计

### 4.1 代码生成流程

```typescript
// 代码生成服务设计

@Injectable()
export class CodeGeneratorService {
  async generateProject(config: ProjectConfig): Promise<GenerateResult> {
    // 1. 验证配置
    this.validateConfig(config);

    // 2. 创建项目目录
    const projectPath = await this.createProjectDirectory(config);

    // 3. 生成数据模型
    const entities = await this.generateEntities(config.entities);

    // 4. 生成Controller
    const controllers = await this.generateControllers(entities);

    // 5. 生成Service
    const services = await this.generateServices(entities);

    // 6. 生成Repository
    const repositories = await this.generateRepositories(entities);

    // 7. 生成配置文件
    const configs = await this.generateConfigs(config);

    // 8. 保存到文件系统
    await this.saveFiles(projectPath, {
      entities,
      controllers,
      services,
      repositories,
      configs,
    });

    // 9. 记录生成历史
    await this.recordGeneration(config);

    return { success: true, projectPath };
  }
}
```

### 4.2 状态管理设计

```typescript
// Pinia状态管理示例

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Project } from '@shared/types';

export const useProjectStore = defineStore('project', () => {
  // 状态
  const projects = ref<Project[]>([]);
  const currentProject = ref<Project | null>(null);
  const loading = ref(false);

  // 计算属性
  const activeProjects = computed(() => projects.value.filter((p) => p.status === 'active'));

  // Actions
  async function fetchProjects() {
    loading.value = true;
    try {
      const response = await projectService.getProjects();
      projects.value = response.data;
    } finally {
      loading.value = false;
    }
  }

  async function createProject(config: ProjectConfig) {
    const response = await projectService.create(config);
    projects.value.push(response.data);
    return response.data;
  }

  async function deleteProject(id: string) {
    await projectService.delete(id);
    projects.value = projects.value.filter((p) => p.id !== id);
  }

  return {
    projects,
    currentProject,
    loading,
    activeProjects,
    fetchProjects,
    createProject,
    deleteProject,
  };
});
```

## 5. 通信机制

### 5.1 IPC通信

```typescript
// Electron主进程IPC处理器

import { ipcMain } from 'electron';

export function registerIPCHandlers() {
  // 生成代码
  ipcMain.handle('generate-code', async (event, prompt, options) => {
    try {
      const response = await axios.post('http://localhost:3001/api/code/generate', {
        prompt,
        options,
      });
      return { success: true, data: response.data };
    } catch (error) {
      return { success: false, error: error.message };
    }
  });

  // 选择目录
  ipcMain.handle('select-directory', async () => {
    const result = await dialog.showOpenDialog({
      properties: ['openDirectory', 'createDirectory'],
    });
    return result.filePaths[0] || null;
  });
}
```

### 5.2 WebSocket通信

```typescript
// NestJS WebSocket Gateway

@WebSocketGateway(3002, { cors: true })
export class CodeGenerationGateway implements OnGatewayConnection {
  @WebSocketServer()
  server: Server;

  handleConnection(client: Socket) {
    console.log('Client connected:', client.id);
  }

  @SubscribeMessage('generate')
  async handleGenerate(@MessageBody() data: GenerateRequest, @ConnectedSocket() client: Socket) {
    // 发送进度更新
    client.emit('progress', { status: 'started', progress: 0 });

    // 执行生成
    await this.codeGenerator.generate(data);

    client.emit('progress', { status: 'completed', progress: 100 });
  }
}
```
